cmake_minimum_required(VERSION 3.16)
project(mapf_ros CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(mapf_ros_msgs REQUIRED)
find_package(mapf_exec REQUIRED)
find_package(gazebo_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)

add_executable(cell_follower src/cell_follower.cpp)
target_link_libraries(cell_follower mapf_exec::mapf_exec)
ament_target_dependencies(cell_follower rclcpp geometry_msgs nav_msgs sensor_msgs mapf_ros_msgs)
add_executable(mapf_executive src/mapf_executive.cpp)
target_link_libraries(mapf_executive mapf_exec::mapf_exec nlohmann_json::nlohmann_json)
ament_target_dependencies(mapf_executive rclcpp nav_msgs gazebo_msgs mapf_ros_msgs)

install(TARGETS cell_follower mapf_executive DESTINATION lib/${PROJECT_NAME})
ament_package()
